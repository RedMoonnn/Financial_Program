### 2. 爬虫性能瓶颈 ⚠️

**问题 (backend/crawler/crawler.py:463-530)：**

- 使用同步 requests 库，每个请求阻塞线程
- 串行采集 41 个组合（8个市场×4个周期 + 3个板块×3个周期），耗时长
- 每 5 分钟全量采集，服务器压力大

**性能对比：**

| 方案             | 采集时间 | 并发数 | 代码复杂度 |
|------------------|----------|--------|------------|
| 当前（同步）     | ~82秒    | 1      | 低         |
| aiohttp（异步）  | ~12秒    | 10     | 中         |
| aiohttp + 连接池 | ~8秒     | 20     | 中         |

**改进方案 (参考 async-python-patterns skill)：**

```python
# crawler/async_crawler.py
import asyncio
import aiohttp
from typing import List, Dict

async def fetch_flow_data_async(
    session: aiohttp.ClientSession,
    flow_type: str,
    market_type: str,
    period: str,
    **kwargs
) -> List[Dict]:
    """异步爬取单个组合的数据"""
    params = build_params(flow_type, market_type, period, **kwargs)

    async with session.get(BASE_URL, headers=HEADERS, params=params) as response:
        data = await response.text()
        return parse_response(data, flow_type, market_type, period)

async def crawl_all_data_async() -> Dict:
    """异步并发爬取所有数据"""
    connector = aiohttp.TCPConnector(limit=20, limit_per_host=5)
    timeout = aiohttp.ClientTimeout(total=30)

    async with aiohttp.ClientSession(connector=connector, timeout=timeout) as session:
        tasks = []

        # 个股资金流
        for market_choice in range(1, 9):
            for day_choice in range(1, 5):
                task = fetch_flow_data_async(
                    session, "Stock_Flow",
                    market_names[market_choice-1],
                    periods[day_choice-1],
                    market_choice=market_choice
                )
                tasks.append(task)

        # 板块资金流
        for detail_choice in range(1, 4):
            for day_choice in range(1, 4):
                task = fetch_flow_data_async(
                    session, "Sector_Flow",
                    detail_flows_names[detail_choice-1],
                    periods[day_choice-1],
                    detail_choice=detail_choice
                )
                tasks.append(task)

        # 并发执行所有任务
        results = await asyncio.gather(*tasks, return_exceptions=True)

        # 过滤错误并入库
        successful = [r for r in results if not isinstance(r, Exception)]
        return {"total": len(successful), "crawl_time": get_now()}

# 在 FastAPI 中使用
async def start_crawler_job_async():
    """异步爬虫任务"""
    while True:
        try:
            set_data_ready(False)
            result = await crawl_all_data_async()
            set_data_ready(True)
            print(f"采集完成：{result}")
        except Exception as e:
            print(f"采集失败：{e}")

        await asyncio.sleep(300)  # 等待5分钟
```

**预期提升：**

- 采集时间从 82秒 → 8-12秒（约 7-10倍提速）
- 减少 CPU 阻塞时间
- 提高服务响应性

from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession
from sqlalchemy.orm import sessionmaker

# 创建异步引擎
async_engine = create_async_engine(
    "mysql+aiomysql://user:pass@host/db",
    pool_size=20,
    max_overflow=10,
    pool_pre_ping=True,
    echo=False

    async_engine,
    class_=AsyncSession,

#### 方案 B：优化批量插入

```python
# crawler/crawler.py:store_data_to_db
def store_data_to_db_optimized(data, table_name):
    """使用 executemany 批量插入"""
    if not data:
        return

    db_config = get_db_config()
    conn = pymysql.connect(**db_config)
    cursor = conn.cursor()

    # 建表...
    cursor.execute(f"TRUNCATE TABLE `{table_name}`;")

    # 批量插入（一次性提交所有记录）
    insert_sql = f"""
        INSERT INTO `{table_name}` (
            `code`, `name`, `flow_type`, ...
        ) VALUES (
            %(code)s, %(name)s, %(flow_type)s, ...
        )
    """
    cursor.executemany(insert_sql, data)  # 使用 executemany

    conn.commit()
    cursor.close()
    conn.close()
```

**预期提升：**

- 批量插入性能提升 3-5倍
- 减少数据库连接开销
- 支持高并发场景

### 5. 异步代码不规范

**问题 (api/v1/endpoints/ai.py:17)：**

- generate_stream_response 是 async 函数，但内部调用的 DeepseekAgent.analyze_stream 是同步生成器
- 混用同步和异步代码，容易引发阻塞

**当前代码：**

```python
async def generate_stream_response(...):
    """异步生成器"""
    stream = DeepseekAgent.analyze_stream(...)  # ❌ 同步调用
    for chunk in stream:  # ❌ 同步迭代
        yield f"data: {data}\n\n"
```

**改进方案 (参考 async-python-patterns skill)：**

```python
# services/ai/deepseek.py
async def analyze_stream_async(...):
    """真正的异步流式分析"""
    client = AsyncOpenAI(api_key=DEEPSEEK_API_KEY, base_url=DEEPSEEK_BASE_URL)

    try:
        stream = await client.chat.completions.create(
            model="deepseek-reasoner",
            messages=[...],
            stream=True
        )

        async for chunk in stream:  # ✅ 异步迭代
            if chunk.choices and chunk.choices[0].delta:
                delta = chunk.choices[0].delta
                if delta.reasoning_content:
                    yield {"type": "thinking", "content": delta.reasoning_content}
                if delta.content:
                    yield {"type": "text", "content": delta.content}
    except Exception as e:
        yield {"type": "error", "content": str(e)}

# api/v1/endpoints/ai.py
async def generate_stream_response(...):
    """异步生成器"""
    async for chunk in DeepseekAgent.analyze_stream_async(...):  # ✅ 异步迭代
        data = json.dumps(chunk, ensure_ascii=False)
        yield f"data: {data}\n\n"
    yield "data: [DONE]\n\n"
```
